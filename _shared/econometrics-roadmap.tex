%   \roadmap{}        the same map with nothing marked (no deck uses this

%
% Weeks before today's are greyed, today's is a green pill, later ones are set
% in ink. Edit \rmitem lines to change titles; edit \rmregister if you add or
% remove a week. The labels here must match the lecture titles and the section
% names in the closing summary — the same three names in all three places.
%
% Numbering is block.week: 1.x foundations, 2.x simple regression,
% 3.x multiple regression, 4.x limits.
%
% FOUR columns rather than ECON42550's three, so each column is narrower and a
% label that wraps is visible. Every box is \raggedright: a \parbox justifies
% by default, which stretches a wrapped line across its column and is what made
% the first draft of this map look broken.
% ---------------------------------------------------------------------------

\makeatletter

% --- position register -----------------------------------------------------
\newcount\rm@pos
\newcount\rm@cur
\newcommand{\rmregister}[1]{\global\advance\rm@pos by 1
  \expandafter\xdef\csname rm@ix@#1\endcsname{\the\rm@pos}}

\rm@pos=0
\rmregister{1.1} \rmregister{1.2} \rmregister{1.3} \rmregister{2.1} \rmregister{2.2} \rmregister{2.3} \rmregister{2.4} \rmregister{3.1} \rmregister{3.2} \rmregister{3.3} \rmregister{4.1} \rmregister{4.2}

% --- one week line ---------------------------------------------------------
\newcount\rm@this
\newcommand{\rmitem}[2]{%
  \rm@this=\@nameuse{rm@ix@#1}\relax
  \ifnum\rm@this=\rm@cur
    \colorbox{dgGreen}{%
      \parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
        \raggedright\color{white}\scriptsize\bfseries #1\hspace{0.45em}#2}}%
  \else
    \ifnum\rm@this<\rm@cur
      \parbox{\linewidth}{\raggedright\color{dgMuted}\scriptsize #1\hspace{0.45em}#2}%
    \else
      \parbox{\linewidth}{\raggedright\color{dgInk}\scriptsize #1\hspace{0.45em}#2}%
    \fi
  \fi
  \par\vspace{0.30em}}

% --- one block column ------------------------------------------------------
\newcommand{\rmhead}[1]{%
  \parbox{\linewidth}{\raggedright\color{dgNavy}\footnotesize\bfseries #1}\par
  \vspace{0.25em}{\color{dgRule}\rule{\linewidth}{0.4pt}}\par\vspace{0.45em}}

% --- the map ---------------------------------------------------------------
